% minimal_slides.tex
%
% Smallest runnable accessible-slides example. Compile with:
%
%   lualatex minimal_slides.tex
%
% Once this works, see accessible_slides.tex for the fully commented
% teaching version with section separators, custom footers, bibliography
% splitting, and validator notes.

\DocumentMetadata{
  lang        = en-US,
  pdfstandard = ua-2,
  pdfstandard = a-4,
  tagging     = on,
  tagging-setup = {math/setup = mathml-SE},
}

\documentclass[frame-title-arg]{ltx-talk}

\usepackage{graphicx}
\usepackage{xcolor}

% Tag frame titles as H1 so screen readers see them as headings.
\tagpdfsetup{role/new-tag = frametitle/H1}

% NOTE: do NOT add \tagpdfsetup{math/alt/use} here. It attaches an /Alt
% string (the raw LaTeX source) to every math Formula tag, and a screen
% reader reads that /Alt *instead of* descending into the generated
% MathML -- so it silences a Blackboard Ally false positive at the cost
% of hiding the real math semantics from screen readers. The MathML
% (math/setup=mathml-SE, set in \DocumentMetadata) is what carries the
% accessible math. See VALIDATION.md.

% Note: ltx-talk's \maketitle currently interacts badly with the
% frame-title tagging machinery in the default TeX Live 2026
% configuration, so we build a tiny title slide manually instead
% (matching the pattern used in accessible_slides.tex).

\title{A minimal accessible slide deck}
\author{Your name}

\begin{document}

\begin{frame*}{}
  \centering
  \Large A minimal accessible slide deck\par
  \vspace{1em}
  \normalsize Your name\par
  \today
\end{frame*}

\begin{frame}{Hello, accessible world}
  \begin{itemize}
    \item Compiled with LuaLaTeX.
    \item Tagging is on; math becomes MathML automatically: \(E = mc^2\).
    \item Every image needs an alt description:
  \end{itemize}
  \begin{center}
    \includegraphics[height=2cm, alt={A capybara's head facing left}]{capybara.jpg}
  \end{center}
\end{frame}

\begin{frame}{One accessible table}
  \tagpdfsetup{table/header-rows = {1}}
  \begin{tabular}{lr}
    \textbf{Item} & \textbf{Count} \\
    Apples        & 3              \\
    Oranges       & 5              \\
  \end{tabular}
\end{frame}

\end{document}
