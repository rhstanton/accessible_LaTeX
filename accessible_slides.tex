% REQUIRED: this block must come before \documentclass.

% - Change lang=en-US if your slides are not in U.S. English.

% USUALLY LEAVE UNCHANGED:

% - These settings target PDF/UA-2 + PDF/A-4.
% - a-4f is only useful if you intentionally embed associated files AND those
%   files are themselves PDF/A compliant. The helper files the LaTeX tagging

%   (no embedded files) is the right default here.

  lang        = en-US,
  pdfstandard = ua-2,
  pdfstandard = a-4,
  tagging=on,
  tagging-setup={math/setup=mathml-SE}
}

%
% Accessibility rule of thumb:
% - Normal text should have at least a 4.5:1 contrast ratio against the

%%% * PART G. PDF TAGGING AND BOOKMARKS
%
% Important distinction:
% - PDF tagging is required for accessibility and screen readers.
% - PDF bookmarks are optional navigation entries in the PDF side pane.
%
% This section does both:
% 1. Tags frame titles as H1 headings.
% 2. Adds section and frame titles to the PDF bookmarks.

% --------------------------------------------------------------------------
% G1. Required tagging adjustment for ltx-talk section titles
% --------------------------------------------------------------------------
%
% ltx-talk uses \section for slide organization. If sections are also tagged as
% headings, some validators report heading-order warnings. This template tags
% frame titles as H1 headings instead.

\ExplSyntaxOn
\AtBeginDocument{
  \NewTaggingSocketPlug { talk / sec / title } { none } { }
  \AssignTaggingSocketPlug { talk / sec / title } { none }
}
\ExplSyntaxOff

\tagpdfsetup{role / new-tag = frametitle / H1}

% --------------------------------------------------------------------------
% G2. Optional PDF bookmarks
% --------------------------------------------------------------------------
%
% OPTIONAL: remove this subsection if you do not want PDF bookmarks.
%
% The code suppresses hyperref's default section bookmark and writes a cleaner
% custom bookmark. Frame titles become level-2 bookmarks beneath sections.

\makeatletter
\DeclareCommandCopy{\OriginalSectionForBookmarks}{\section}

\RenewDocumentCommand{\section}{s o m}{%
  \begingroup
    \def\Hy@writebookmark##1##2##3##4##5{}%
    \IfBooleanTF{#1}
      {\phantomsection\IfNoValueTF{#2}{\OriginalSectionForBookmarks*{#3}}{\OriginalSectionForBookmarks*[#2]{#3}}}%
      {\IfNoValueTF{#2}{\OriginalSectionForBookmarks{#3}}{\OriginalSectionForBookmarks[#2]{#3}}}%
  \endgroup
  \IfBooleanTF{#1}
    {\IfNoValueTF{#2}{\pdfbookmark[1]{#3}{section-star.\thepage.\number\inputlineno}}{\pdfbookmark[1]{#2}{section-star.\thepage.\number\inputlineno}}}%
    {\IfNoValueTF{#2}{\pdfbookmark[1]{Section \thesection: #3}{section.\thesection}}{\pdfbookmark[1]{Section \thesection: #2}{section.\thesection}}}%
}
\makeatother

\ExplSyntaxOn
\int_new:N \g_accessibleslides_frame_bookmark_int

\AddToHook{cmd/frametitle/after}{%
  \tl_if_empty:NF \g__talk_frame_title_tl
    {
      \int_gincr:N \g_accessibleslides_frame_bookmark_int
      \pdfbookmark[2]{\tl_use:N \g__talk_frame_title_tl}
        {frame.\int_use:N \g_accessibleslides_frame_bookmark_int}
    }
}
\ExplSyntaxOff

% Helper: turn an ordinary macro body into an ltx-talk footer element.
\newcommand{\DeclareTalkFooterElement}[2]{%
  \expandafter\def\csname @#1\endcsname{#2}%
  \DeclareInstance{footer-element}{#1}{talk}{}%
}

% Regular slide footer: right-aligned frame number.
\DeclareTalkFooterElement{RegularSlideFooter}{%
  \makebox[\dimexpr\paperwidth-1.2cm\relax][r]{\arabic{frame}}%
}

% Title slide footer: logo left, copyright centered.
\DeclareTalkFooterElement{TitleSlideFooter}{%
  \makebox[\dimexpr\paperwidth-0.6cm\relax][l]{%
    \rlap{\raisebox{0pt}[.25in][0pt]{%
      \includegraphics[width=.45in,alt={Berkeley Haas logo}]{BerkeleyHaas}}}%
    \hfill
    \textcopyright\ \TalkAuthor, \the\year
    \hfill
    \phantom{\arabic{frame}}%
  }%
}

% Regular page style used by ordinary slides.
\EditInstance{footer}{std}{
  color         = black,
  font          = \tiny,
  left-hspace   = 0.9cm,
  right-hspace  = 0.3cm,
  element-order = {RegularSlideFooter},
}

% Title-page-only footer style.
\DeclareInstanceCopy{footer}{TitleSlideFooterStyle}{std}
\EditInstance{footer}{TitleSlideFooterStyle}{
  left-hspace   = 0.3cm,
  right-hspace  = 0.3cm,
  element-order = {TitleSlideFooter},
}

\def\ps@AccessibleTitlePage{%

  \let\@evenhead\@oddhead
  \let\@evenfoot\@oddfoot
}

\begin{frame}{\LaTeX\ and accessibility: new requirements}

\begin{itemize}

\item From \BlueText{May 11, 2026}, course materials must meet \hrefu{https://www.w3.org/WAI/standards-guidelines/wcag/}{WCAG} 2.1 Level AA accessibility standards
\begin{itemize}
\item Applies even to password-protected course sites (e.g., \BlueText{bCourses}).
%\item Campus guidance issued Jan 8, 2026.
\end{itemize}

\bigskip

\item Many of us use \BlueText{\LaTeX} to create teaching materials, both slides and documents.

\bigskip
  
\item \textbf{\RedText{But standard \LaTeX\ (especially \BlueText{Beamer}) does not automatically generate accessible PDFs.}}
  
\bigskip  
  
\item \GreenText{Good news:} The \LaTeX\ Project team has developed solutions.
\end{itemize}    
  
\end{frame}

\begin{frame}{What is this project?}

\begin{itemize}
\item A practical template set for accessible \LaTeX\ using the \hrefu{https://latex3.github.io/tagging-project/}{\LaTeX\ Tagging Project}
  \begin{itemize}
  \item \GreenText{Two templates:} Articles (\RedText{\texttt{article}} class) and slides (\RedText{\texttt{ltx-talk}})
  \item \GreenText{Shows:} What is shared and what differs
  \item \GreenText{Contains:} Math, text, figures, and tables
  \item \GreenText{Validation status:} Slides score 100\% with \BlueText{Ally} and pass common validators (\BlueText{veraPDF}, \BlueText{PAC}, \BlueText{Acrobat}) in typical workflows (tested March 17, 2026).
  \end{itemize}

  \bigskip

\item Also includes some (optional) additional slide features
  \begin{itemize}
  \item Readable  \BlueText{outline slide}
  \item Automatic \BlueText{section separator slides}
  \item Multi-page \BlueText{bibliography support}
  \end{itemize}
  
\bigskip

\item Repo: \urlu{https://github.com/rhstanton/accessible_LaTeX}

\end{itemize}
\end{frame}

\begin{frame}{Converting to accessible \LaTeX: the big picture}
\begin{itemize}
\item \GreenText{Your existing \LaTeX\ skills transfer:} mainly adding a few consistent accessibility features.
\item \GreenText{Most changes are local:} metadata once; then per-figure and per-table tweaks.

\bigskip

\item \textbf{Applies to both slides \& articles:}
  \begin{itemize}
  \item Document metadata + PDF tagging
  \item Alt text on images
  \item Table header row tagging
  \item Sufficient color contrast
  \item Compile with Lua\LaTeX
  \end{itemize}

\bigskip

\item \textbf{Key difference:}
  \begin{itemize}
  \item Articles: \GreenText{keep your existing class} and add tagging features
  \item Slides: switch to \texttt{ltx-talk} (frame syntax similar to Beamer), then recreate styling
  \end{itemize}
\end{itemize}
\end{frame}

\section{Setup}

\begin{frame}{Version requirements: what actually matters}
\begin{itemize}
\item \GreenText{Engine:} must compile with \BlueText{Lua\LaTeX} (not pdf\LaTeX\ or Xe\LaTeX).

\bigskip

\item \GreenText{LaTeX kernel/format date:} must be \BlueText{2025-11-01 or later}.
\item \textbf{Important:} a “TeX Live year” alone is not a reliable guarantee.
  \begin{itemize}
  \item You need a sufficiently new \LaTeX\ kernel \emph{on your install}, or use Overleaf Labs.
  \end{itemize}

\bigskip

\item \textbf{Quick check (in any document preamble):}

  \medskip  
\fbox{%
\begin{minipage}{0.92\linewidth}
\ttfamily\small
% prints your LaTeX format date in the .log\\
\textbackslash show\textbackslash fmtversion
\end{minipage}}
\end{itemize}
\end{frame}

\begin{itemize}

  \begin{enumerate}

\section{Core requirements}

\begin{frame}{Core requirement 1: document metadata}

\smallskip

\begin{itemize}
\small
\item \BlueText{\textbf{Required for both slides and articles}}

% \bigskip

\item Every accessible document \RedText{must} begin with \texttt{\textbackslash DocumentMetadata}
\item Goes \RedText{before} \texttt{\textbackslash documentclass}

\bigskip

\fbox{%
\begin{minipage}{0.9\linewidth}
\ttfamily\scriptsize
\textbackslash DocumentMetadata\{\\[0.5ex]
\hspace*{0.2in}\% Request both PDF/A archival conformance and PDF/UA accessibility tagging\\[0.5ex]
\hspace*{0.2in}pdfstandard=a-4,\hspace*{0.49in}\% PDF/A-4 format (use a-4f only if embedding files)\\[0.5ex]
\hspace*{0.2in}pdfstandard=ua-2,\hspace*{0.42in}\% Add PDF/UA-2 conformance target\\[0.5ex]
\hspace*{0.2in}lang=en-US,\hspace*{0.78in}\% Language for screen readers\\[0.5ex]
\hspace*{0.2in}tagging=on,\hspace*{0.78in}\% Enable PDF tagging\\[0.5ex]
\hspace*{0.2in}tagging-setup=\{\\[0.5ex]
\hspace*{0.4in}math/setup=mathml-SE\hspace*{0.22in}\% Enable semantic MathML generation\\[0.5ex]
\hspace*{0.2in}\}\\[0.5ex]
\}
\end{minipage}
}

\bigskip

\item \RedText{\textbf{Why PDF/UA-2 + PDF/A-4?}} This aligns with current ISO standards and modern tagging output.
  % \begin{itemize}
  % \item See \hrefu{https://pdfa.org/wp-content/uploads/2025/05/PDFAdays-2025-talk-tagged.pdf}{PDF Days 2025 (LaTeX Tagged PDF talk)}
  % \end{itemize}    
  % \item Configures accessibility, math tagging, and validator-compatibility behavior in one place.
  
\end{itemize}
\end{frame}

\begin{frame}{Core requirements 2--4: images, tables, and color contrast}
\begin{itemize}
\item \BlueText{\textbf{Required for both slides and articles.}}

\bigskip

\item \textbf{Images:} every \texttt{\textbackslash includegraphics} needs \texttt{alt=\{...\}}.
\item \textbf{Tables:} mark header rows using \texttt{\textbackslash tagpdfsetup\{table/header-rows=\{...\}\}}.
\item \textbf{Colors:} WCAG 2.1 AA requires \BlueText{4.5:1} contrast ratio for normal text.
\bigskip

\item \textbf{Accessible color defaults:}

\medskip  
  
\fbox{%
\begin{minipage}{0.92\linewidth}
\ttfamily
\textbackslash colorlet\{AccessibleRed\}\{red!80!black\}\\
\textbackslash colorlet\{AccessibleGreen\}\{green!40!black\}\\
\% Standard blue is fine
\end{minipage}}
\end{itemize}
\end{frame}

\begin{frame}{Core requirement 5: compile with Lua\LaTeX}
\begin{itemize}
\item \BlueText{\textbf{Required for both slides and articles.}}
\item Compile with \RedText{Lua\LaTeX} (not pdf\LaTeX\ or Xe\LaTeX).

\bigskip

\item \textbf{Why Lua\LaTeX?}
  \begin{enumerate}
  \item Automatic math accessibility support (no manual MathML files in normal workflows)
  \item Full tagging support
  \item Good Unicode/OpenType font support
  \end{enumerate}

\bigskip

\item \textbf{How to switch:}
  \begin{itemize}
  \item Command line: \texttt{lualatex filename.tex}
  \item Editors: select \texttt{LuaLaTeX} as the compiler
  \end{itemize}
\end{itemize}
\end{frame}

\section{Examples}

\begin{frame}{Slide content: mostly the same as Beamer}
\begin{itemize}
\item Slides live in \RedText{\texttt{frame}} environments (similar to Beamer).
\item So your content often transfers with minimal changes.

\bigskip

\item \textbf{Typical frame skeleton:}

  \medskip
  
\fbox{%
\begin{minipage}{0.92\linewidth}
\ttfamily
\textbackslash begin\{frame\}\{Title\}\\
\textbackslash begin\{itemize\}\\
\hspace*{0.2in}\textbackslash item\ First point\\
\textbackslash end\{itemize\}\\
\textbackslash end\{frame\}
\end{minipage}}

\bigskip

\item \BlueText{Note (optional):} template fonts keep digits/percent/dollar signs consistent in math/text.
  \begin{itemize}
  \item \makebox[2.75em][l]{\GreenText{Text:}}\ $\$1234567890\%.$
  \item \makebox[2.75em][l]{\GreenText{Math:}}\ $\$1234567890\%.$
  \end{itemize}
\end{itemize}

\end{frame}

\begin{frame*}{Figures: alt text}
\begin{itemize}
\item Figures are included as usual (e.g., \texttt{\textbackslash includegraphics}), but \RedText{add alt text}.

\bigskip

\item \textbf{Good (minimal):}
\fbox{\ttfamily
  \textbackslash includegraphics[\AltBox{alt=\{A capybara\}}]\{capybara.jpg\}
}

\bigskip

\item \textbf{Better (more informative):}

  \medskip  
\fbox{\ttfamily
\textbackslash includegraphics[alt=\{A capybara's head facing left\}]\{capybara.jpg\}}

\bigskip

\item Decorative images: use empty alt text: \texttt{alt=\{\}}.
\end{itemize}

\vspace*{0.15in}
\centering
\includegraphics[height=.42\textheight, alt={A capybara's head facing left}]{capybara.jpg}
\end{frame*}

\small

\section{Slides versus articles}

\begin{frame}{Articles vs.\ slides}
\begin{itemize}
\item \textbf{Articles:} \GreenText{keep your existing class} (e.g., \texttt{article}) and add the checklist items.
\item \textbf{Slides:} switch to \texttt{ltx-talk}.
  \begin{itemize}
  \item Your \texttt{\textbackslash begin\{frame\}} content often stays the same.
  \item Remove Beamer themes/templates; recreate styling with standard \LaTeX\ packages (as in this template).
  \end{itemize}

\bigskip

\item \textbf{If you’re converting Beamer slides:}

\medskip  
\fbox{%
\begin{minipage}{0.92\linewidth}
\ttfamily
\AltBox{\textbackslash documentclass\{ltx-talk\}} \quad \textrm{(instead of beamer)}
\end{minipage}}
\end{itemize}
\end{frame}

\begin{frame}{Getting started (new vs.\ converting): keep it simple}
\begin{itemize}
\item \textbf{New documents:}
  \begin{itemize}
  \item copy \texttt{accessible\_slides.tex} or \texttt{accessible\_article.tex}
  \item replace content; keep the metadata/preamble structure
  \end{itemize}

\bigskip

\item \textbf{Converting existing documents:}
  \begin{itemize}
  \item add metadata + Lua\LaTeX\ + alt text + table header rows
  \item for slides, switch to \texttt{ltx-talk} and redo styling once
  \end{itemize}

\end{itemize}
\end{frame}

\section{Validation and known issues}

\begin{frame}{Common pitfalls}

\begin{itemize}
\item \RedText{Confusing PDF tagging with PDF bookmarks}
  \begin{itemize}
  \item \texttt{tagging=on} creates the structure tree for \GreenText{screen readers} (required).
  \item PDF bookmarks (navigation pane) are separate and optional.
  \end{itemize}

\item \RedText{Forgetting alt text}
  \begin{itemize}
  \item Every \texttt{\textbackslash includegraphics} needs \texttt{alt=\{...\}} (or \texttt{alt=\{\}} if decorative).
  \end{itemize}

\item \RedText{Not tagging table header rows}
  \begin{itemize}
  \item Add \texttt{\textbackslash tagpdfsetup\{table/header-rows=\{...\}\}} before each table.
  \end{itemize}

\item \RedText{Low contrast colors}
  \begin{itemize}
  \item Avoid light colors (e.g., plain yellow/cyan).
  \item Use darker variants for red/green (defined in this template).
  \end{itemize}

\item \RedText{Wrong engine or old kernel}
  \begin{itemize}
  \item Use Lua\LaTeX\ and ensure format date \texttt{2025-11-01} or later.
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Validation workflow: use multiple checkers}
\begin{itemize}
\item No single checker catches everything.
\item Accessibility usability checks and strict PDF conformance checks are not identical.
\item Practical workflow: run \BlueText{2+} tools and compare findings.

\bigskip

\item \textbf{Common tools:}
  \begin{itemize}
  \item \BlueText{Ally} (bCourses)
  \item \BlueText{veraPDF} (PDF/A conformance): \urlu{https://verapdf.org/}
  \item \BlueText{PAC} (PDF accessibility): \urlu{https://pac.pdf-accessibility.org/}
  \item \BlueText{Adobe Acrobat} (useful but has known quirks)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Optional cleanup for strict validators: \texttt{strip\_af.py}}
\begin{itemize}
\item The metadata settings alone can be enough for a 100\% score in Ally.
\item Some strict tools (e.g., \BlueText{veraPDF}) may still complain about \texttt{/AF} or embedded-file structures.
\item To remove these, run:

\bigskip

\fbox{%
\begin{minipage}{0.92\linewidth}
\ttfamily
python strip\_af.py build/accessible\_slides.pdf
\end{minipage}
}

\bigskip

\item Prerequisite: install the Python package \GreenText{pikepdf} first: \texttt{python -m pip install pikepdf}.
\item Then validate again with veraPDF.
\end{itemize}
\end{frame}

\begin{frame}{Validator-specific notes (common false positives)}

\begin{itemize}
\item \textbf{Acrobat: ``Lbl and LBody -- Failed'' (validator bug)}
  \begin{itemize}
  \item Often appears around figure captions, section numbers, table captions.
  \item \GreenText{This is an Acrobat bug} in common setups; typically safe to ignore if source tagging is correct.
  \item Reference: \urlu{https://tex.stackexchange.com/a/709655/2388}
  \end{itemize}

\bigskip

\item \textbf{PAC table warning (known behavior)}
  \begin{itemize}
  \item PAC can report: \texttt{Table header cell has no associated subcells} even when \texttt{table/header-rows} is correct.
  \item Documented by the Tagging Project: \urlu{https://github.com/latex3/tagging-project/issues/1056}
  \end{itemize}

\bigskip

\item \textbf{Bottom line:} treat one-tool warnings as investigation prompts, not definitive errors.
\end{itemize}
\end{frame}

\section{Quick start}

\begin{frame}{Quick start: accessible \LaTeX\ on one slide}
\begin{itemize}
\item \textbf{To make an accessible \LaTeX\ PDF:}
  \begin{enumerate}
  \item Add \texttt{\textbackslash DocumentMetadata\{...\}} (before \texttt{\textbackslash documentclass})
  \item Compile with \BlueText{Lua\LaTeX}
  \item Every figure: \texttt{\textbackslash includegraphics[alt=\{...\}]\{...\}}
  \item Every table: \texttt{\textbackslash tagpdfsetup\{table/header-rows=\{...\}\}}
  \item Use high-contrast colors (4.5:1) and validate with 2+ tools
  \end{enumerate}

\section{(Optional) additional utilities}

\begin{frame}{Optional presentation utilities}

\begin{itemize}
\item This template includes \GreenText{optional utilities} in \texttt{slide\_utils.sty}
  \begin{itemize}
  \item \RedText{Not required for accessibility}
  \item Useful for common presentation tasks
  \item Easy to customize or remove
  \end{itemize}

\bigskip

\item \textbf{Two optional features:}
  \begin{enumerate}
  \item Automatic section separator slides
  \item Multi-page bibliography support
  \end{enumerate}

\bigskip

\item See \texttt{slide\_utils.sty} for implementation details and customization options.
\item \textbf{Note:} There's additional code in the main preamble to get readable table of contents.
  \end{itemize}
\end{frame}

\begin{frame}{Optional feature 1: section separator slides}

\begin{itemize}
\item \textbf{What it does:} Automatically inserts a separator slide after each \texttt{\textbackslash section} command
  \begin{itemize}
  \item Displays section number and title in large text
  \item Centered on the slide for visual emphasis
  \item Uses accessible colors (AccessibleRed for sections, blue for subsections)
  \end{itemize}

\bigskip

\item \textbf{Example:} After \texttt{\textbackslash section\{Core requirements\}}, automatically creates a slide saying

  \bigskip
  \begin{center}
  {\color{AccessibleRed}\Large\bfseries 3. Core requirements}
  \end{center}

\bigskip

\item \textbf{To disable:} Comment out or remove the redefinition code in \texttt{slide\_utils.sty} Section 1
\item \textbf{To customize:} Edit colors, fonts, or spacing in the frame template
\end{itemize}
\end{frame}

\begin{frame}{Optional feature 2: multi-page bibliography}

  \medskip
  
\begin{itemize}
\item \textbf{\RedText{Problem:}} \texttt{\textbackslash bibliography} inside \texttt{frame} environment puts all references on one slide.
  \begin{itemize}
  \item For large bibliographies, this can be unreadable.
  \item This was easy with Beamer, using the \texttt{allowframebreaks} option.
  \item But \texttt{ltx-talk} does not have a similar option.
  \end{itemize}

\bigskip

\item \textbf{\GreenText{Solution:}} Write code to split bibliography automatically across multiple slides
  \begin{itemize}
  \item Works with standard Bib\TeX\ workflow
  \item Uses Lua script (\texttt{\BlueText{rsbibsplit.lua}}) to split the \texttt{.bbl} file
  \item (Lua\LaTeX\ provides access to the \BlueText{Lua} programming language)
  \end{itemize}

\bigskip

\item \textbf{Usage:} As usual (see end of this \texttt{.tex} file for complete example)
  \begin{itemize}
  \item \texttt{\textbackslash cite\{key1,key2,...\}} to specify references
  \item \texttt{\textbackslash bibliography\{bibfile\}} to specify \texttt{bib} file
  \item (Set \texttt{\textbackslash RS@bibmax} to customize max per slide; default=8).
  \item \BlueText{\textbf{Note:} Do \textbf{not} put \texttt{\textbackslash bibliography} command inside \texttt{frame} environment }
    
  \end{itemize}

\bigskip

\item See \texttt{slide\_utils.sty} Section 2 for detailed workflow explanation.
\end{itemize}
\end{frame}
