% ======================================================================
% DOCUMENT METADATA - REQUIRED FOR ACCESSIBILITY
% ======================================================================
% This MUST be the first thing in your document, before \documentclass
% It configures the PDF for accessibility compliance
\DocumentMetadata{
  % Request both PDF/A archival conformance and PDF/UA accessibility tagging
  pdfstandard=a-2u,    % PDF/A-2u format (archival standard with Unicode support)
  pdfstandard=ua-1,    % Add PDF/UA-1 accessibility conformance target
  pdfversion=1.7,          % Explicit PDF version used for compatibility with conformance tooling
  lang=en-US,               % Document language (required for screen readers)
  tagging=on,                 % Enable PDF tagging (required for accessibility)
  tagging-setup={
    math/alt/use,                         % Keep math accessibility enabled (screen readers get tagged math)
    math/mathml/AF=false,    % Do not embed MathML as PDF Associated Files (helps some strict conformance checkers)
    math/tex/AF=false,             % Do not embed TeX source as Associated Files
    math/mathml/sources=      % Clear MathML source attachment list (avoid extra embedded helper files)
  }
}

% IMPORTANT DISTINCTION:
% - "tagging = on" in \DocumentMetadata enables PDF TAGGING (structure tree)
%   * This is for screen readers and accessibility compliance
%   * This is REQUIRED for WCAG compliance

%   * These are for sighted users navigating the PDF

% ----------------------------------------------------------------------
% Part 1: PDF Tagging Configuration (Screen Reader Document Structure)
% ----------------------------------------------------------------------
% Controls how document elements are tagged in the PDF structure tree
% This affects how screen readers interpret the document hierarchy
% THIS PART IS REQUIRED FOR ACCESSIBILITY

\ExplSyntaxOn
\AtBeginDocument{
  % Suppress automatic tagging of \section titles
  %
  % WHY: ltx-talk uses \section primarily for organization.
  % Automatic tagging creates empty H1 headings that trigger validator warnings.
  %
  % SOLUTION: Disable section-title tagging; tag frame titles as H1 instead
  \NewTaggingSocketPlug { talk / sec / title } { none } { }
  \AssignTaggingSocketPlug { talk / sec / title } { none }
}
\ExplSyntaxOff

% Treat frame titles as H1 headings in the document structure
\tagpdfsetup{role / new-tag = frametitle / H1}

% ----------------------------------------------------------------------
% Part 2: PDF Bookmark Generation (Navigation Outline)
% ----------------------------------------------------------------------
% Creates PDF bookmarks (outline) for sections and slides
% This affects the navigation pane in PDF readers (not screen reader structure)
%
% NOTE: THIS SECTION IS OPTIONAL AND NOT REQUIRED FOR ACCESSIBILITY!
% It's included here as a convenience feature for navigation.
% The bookmark code below is CUSTOM - it doesn't happen automatically.
% You can remove this entire section if you don't want bookmarks.

% Required packages for bookmark customization
\usepackage{hyperref}   % PDF hyperlinks and bookmarks
\usepackage{bookmark}   % Enhanced bookmark control
\usepackage{xparse}     % Advanced command definitions

% Add section titles to PDF bookmarks (level 1)
\makeatletter
\DeclareCommandCopy{\origsection}{\section}

\RenewDocumentCommand{\section}{s o m}
  {%
    % Call original \section but suppress hyperref's default bookmark
    \begingroup
      \let\rs@Hy@writebookmark\Hy@writebookmark
      \def\Hy@writebookmark##1##2##3##4##5{}%
      \IfBooleanTF{#1}
        {%  Starred version: \section*{...}
          \phantomsection
          \IfNoValueTF{#2}
            {\origsection*{#3}}
            {\origsection*[#2]{#3}}%
        }
        {%  Normal version: \section{...}
          \IfNoValueTF{#2}
            {\origsection{#3}}
            {\origsection[#2]{#3}}%
        }%
    \endgroup
    % Add custom bookmark with desired formatting
    \IfBooleanTF{#1}
      {%  Starred: bookmark uses title as-is
        \IfNoValueTF{#2}
          {\pdfbookmark[1]{#3}{section*.\thepage.\number\inputlineno}}
          {\pdfbookmark[1]{#2}{section*.\thepage.\number\inputlineno}}%
      }
      {%  Normal: prefix with "Section N:"
        \IfNoValueTF{#2}
          {\pdfbookmark[1]{Section \thesection: #3}{section.\thesection}}
          {\pdfbookmark[1]{Section \thesection: #2}{section.\thesection}}%
      }%
  }
\makeatother

\begin{frame}{\LaTeX\ and accessibility: new requirements}

\begin{itemize}

\item From \tb{April 24, 2026}, course materials must meet \hrefu{https://www.w3.org/WAI/standards-guidelines/wcag/}{WCAG} 2.1 Level AA accessibility standards
\begin{itemize}
\item Applies even to password-protected course sites (e.g., \tb{bCourses}).
%\item Campus guidance issued Jan 8, 2026.
\end{itemize}

\bigskip

\item Many of us use \tb{\LaTeX} to create teaching materials, both slides and documents.

\bigskip
  
\item \textbf{\tr{But standard \LaTeX\ (especially \tb{Beamer}) does not automatically generate accessible PDFs.}}
  
\bigskip  
  
\item \tg{Good news:} The \LaTeX\ Project team has developed solutions.
\end{itemize}    
  
\end{frame}

\begin{frame}{What is this project?}

\begin{itemize}
\item A practical template set for accessible \LaTeX\ using the \hrefu{https://latex3.github.io/tagging-project/}{\LaTeX\ Tagging Project}
  \begin{itemize}
  \item \tg{Two templates:} Articles (\tr{\texttt{article}} class) and slides (\tr{\texttt{ltx-talk}})
  \item \tg{Shows:} What is shared and what differs
  \item \tg{Contains:} Math, text, figures, and tables
  \item \tg{Validation status:} Slides score 100\% with \tb{Ally} and pass common validators (\tb{veraPDF}, \tb{PAC}, \tb{Acrobat}) in typical workflows (tested March 17, 2026).
  \end{itemize}

  \bigskip

\item Also includes some (optional) additional slide features
  \begin{itemize}
  \item Readable  \tb{outline slide}
  \item Automatic \tb{section separator slides}
  \item Multi-page \tb{bibliography support}
  \end{itemize}
  
\bigskip

\item Repo: \urlu{https://github.com/rhstanton/accessible_LaTeX}

\end{itemize}
\end{frame}

\begin{frame}{Converting to accessible \LaTeX: the big picture}
\begin{itemize}
\item \tg{Your existing \LaTeX\ skills transfer:} mainly adding a few consistent accessibility features.
\item \tg{Most changes are local:} metadata once; then per-figure and per-table tweaks.

\bigskip

\item \textbf{Applies to both slides \& articles:}
  \begin{itemize}
  \item Document metadata + PDF tagging
  \item Alt text on images
  \item Table header row tagging
  \item Sufficient color contrast
  \item Compile with Lua\LaTeX
  \end{itemize}

\bigskip

\item \textbf{Key difference:}
  \begin{itemize}
  \item Articles: \tg{keep your existing class} and add tagging features
  \item Slides: switch to \texttt{ltx-talk} (frame syntax similar to Beamer), then recreate styling
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Tables: header rows}

\small  
\begin{itemize}
\item Tables use \texttt{tabular} as usual, but you must  specify \tr{header rows}.  
\item Use \texttt{\{1\}} for one header row, \texttt{\{1,2\}} for two header rows, etc.
\item \textbf{Note:} \tb{PAC} may still warn about header associations in some cases; see \tb{Validator-specific notes}.
  
\vspace*{0.05in}
    
\fbox{%
\begin{minipage}{0.95\linewidth}
\ttfamily\footnotesize
\AltBox{\textbackslash tagpdfsetup\{table/header-rows=\{1,2,3\}\}} \\
\textbackslash begin\{tabular\}\{ccccrcccr\} \\
\textbackslash toprule \\
\textrm{\tb{\footnotesize $\leftarrow$ 3 header rows}} \\
\textbackslash midrule \\
\textrm{\tg{\footnotesize $\leftarrow$ data rows}} \\
\textbackslash bottomrule \\
\textbackslash end\{tabular\}
\end{minipage}
}

\end{itemize}

\vspace*{0.05in}
  
\begin{center}\scriptsize
\tagpdfsetup{table/header-rows={1,2,3}}
  \begin{tabular}{ccccrcccr}

\section{Slides versus articles}

\begin{frame}{Articles vs.\ slides}
\begin{itemize}
\item \textbf{Articles:} \tg{keep your existing class} (e.g., \texttt{article}) and add the checklist items.
\item \textbf{Slides:} switch to \texttt{ltx-talk}.
  \begin{itemize}
  \item Your \texttt{\textbackslash begin\{frame\}} content often stays the same.
  \item Remove Beamer themes/templates; recreate styling with standard \LaTeX\ packages (as in this template).
  \end{itemize}

\bigskip

\item \textbf{If you’re converting Beamer slides:}

\medskip  
\fbox{%
\begin{minipage}{0.92\linewidth}
\ttfamily
\AltBox{\textbackslash documentclass\{ltx-talk\}} \quad \textrm{(instead of beamer)}
\end{minipage}}
\end{itemize}
\end{frame}

\begin{frame}{Getting started (new vs.\ converting): keep it simple}
\begin{itemize}
\item \textbf{New documents:}
  \begin{itemize}
  \item copy \texttt{accessible\_slides.tex} or \texttt{accessible\_article.tex}
  \item replace content; keep the metadata/preamble structure
  \end{itemize}

\bigskip

\item \textbf{Converting existing documents:}
  \begin{itemize}
  \item add metadata + Lua\LaTeX\ + alt text + table header rows
  \item for slides, switch to \texttt{ltx-talk} and redo styling once
  \end{itemize}

\bigskip

\item Repo: \urlu{https://github.com/rhstanton/accessible_LaTeX}
\end{itemize}
\end{frame}

\section{Validation and known issues}

\begin{frame}{Common pitfalls}

\begin{itemize}
\item \tr{Confusing PDF tagging with PDF bookmarks}
  \begin{itemize}
  \item \texttt{tagging=on} creates the structure tree for \tg{screen readers} (required).
  \item PDF bookmarks (navigation pane) are separate and optional.
  \end{itemize}

\item \tr{Forgetting alt text}
  \begin{itemize}
  \item Every \texttt{\textbackslash includegraphics} needs \texttt{alt=\{...\}} (or \texttt{alt=\{\}} if decorative).
  \end{itemize}

\item \tr{Not tagging table header rows}
  \begin{itemize}
  \item Add \texttt{\textbackslash tagpdfsetup\{table/header-rows=\{...\}\}} before each table.
  \end{itemize}

\item \tr{Low contrast colors}
  \begin{itemize}
  \item Avoid light colors (e.g., plain yellow/cyan).
  \item Use darker variants for red/green (defined in this template).
  \end{itemize}

\item \tr{Wrong engine or old kernel}
  \begin{itemize}
  \item Use Lua\LaTeX\ and ensure format date \texttt{2025-11-01} or later.
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Validation workflow: use multiple checkers}
\begin{itemize}
\item No single checker catches everything.
\item Accessibility usability checks and strict PDF conformance checks are not identical.
\item Practical workflow: run \tb{2+} tools and compare findings.

\bigskip

\item \textbf{Common tools:}
  \begin{itemize}
  \item \tb{Ally} (bCourses)
  \item \tb{veraPDF} (PDF/A conformance): \urlu{https://verapdf.org/}
  \item \tb{PAC} (PDF accessibility): \urlu{https://pac.pdf-accessibility.org/}
  \item \tb{Adobe Acrobat} (useful but has known quirks)
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Optional cleanup for strict validators: \texttt{strip\_af.py}}
\begin{itemize}
\item The metadata settings alone can be enough for a 100\% score in Ally.
\item Some strict tools (e.g., \tb{veraPDF}) may still complain about \texttt{/AF} or embedded-file structures.
\item To remove these, run:

\bigskip

\fbox{%
\begin{minipage}{0.92\linewidth}
\ttfamily
python strip\_af.py build/accessible\_slides.pdf
\end{minipage}
}

\bigskip

\item Prerequisite: install the Python package \tg{pikepdf} first: \texttt{python -m pip install pikepdf}.
\item Then validate again with veraPDF.
\end{itemize}
\end{frame}

\begin{frame}{Validator-specific notes (common false positives)}

\begin{itemize}
\item \textbf{Acrobat: ``Lbl and LBody -- Failed'' (validator bug)}
  \begin{itemize}
  \item Often appears around figure captions, section numbers, table captions.
  \item \tg{This is an Acrobat bug} in common setups; typically safe to ignore if source tagging is correct.
  \item Reference: \urlu{https://tex.stackexchange.com/a/709655/2388}
  \end{itemize}

\bigskip

\item \textbf{PAC table warning (known behavior)}
  \begin{itemize}
  \item PAC can report: \texttt{Table header cell has no associated subcells} even when \texttt{table/header-rows} is correct.
  \item Documented by the Tagging Project: \urlu{https://github.com/latex3/tagging-project/issues/1056}
  \end{itemize}

\bigskip

\item \textbf{Bottom line:} treat one-tool warnings as investigation prompts, not definitive errors.
\end{itemize}
\end{frame}

\section{Quick start}

\begin{frame}{Quick start: accessible \LaTeX\ on one slide}
\begin{itemize}
\item \textbf{To make an accessible \LaTeX\ PDF:}
  \begin{enumerate}
  \item Add \texttt{\textbackslash DocumentMetadata\{...\}} (before \texttt{\textbackslash documentclass})
  \item Compile with \tb{Lua\LaTeX}
  \item Every figure: \texttt{\textbackslash includegraphics[alt=\{...\}]\{...\}}
  \item Every table: \texttt{\textbackslash tagpdfsetup\{table/header-rows=\{...\}\}}
  \item Use high-contrast colors (4.5:1) and validate with 2+ tools
  \end{enumerate}

\bigskip

\item Template repo: \urlu{https://github.com/rhstanton/accessible_LaTeX}
\item Questions: \hrefu{mailto:richard.stanton@berkeley.edu}{richard.stanton@berkeley.edu}
\end{itemize}

\section{(Optional) additional utilities}

\begin{frame}{Optional presentation utilities}

\begin{itemize}
\item This template includes \tg{optional utilities} in \texttt{slide\_utils.sty}
  \begin{itemize}
  \item \tr{Not required for accessibility}
  \item Useful for common presentation tasks
  \item Easy to customize or remove
  \end{itemize}

\bigskip

\item \textbf{Two optional features:}
  \begin{enumerate}
  \item Automatic section separator slides
  \item Multi-page bibliography support
  \end{enumerate}

\bigskip

\item See \texttt{slide\_utils.sty} for implementation details and customization options.
\item \textbf{Note:} There's additional code in the main preamble to get readable table of contents.
  \end{itemize}
\end{frame}

\begin{frame}{Optional feature 1: section separator slides}

\begin{itemize}
\item \textbf{What it does:} Automatically inserts a separator slide after each \texttt{\textbackslash section} command
  \begin{itemize}
  \item Displays section number and title in large text
  \item Centered on the slide for visual emphasis
  \item Uses accessible colors (AccessibleRed for sections, blue for subsections)
  \end{itemize}

\bigskip

\item \textbf{Example:} After \texttt{\textbackslash section\{Core requirements\}}, automatically creates a slide saying

  \bigskip
  \begin{center}
  {\color{AccessibleRed}\Large\bfseries 3. Core requirements}
  \end{center}

\bigskip

\item \textbf{To disable:} Comment out or remove the redefinition code in \texttt{slide\_utils.sty} Section 1
\item \textbf{To customize:} Edit colors, fonts, or spacing in the frame template
\end{itemize}
\end{frame}

\begin{frame}{Optional feature 2: multi-page bibliography}

  \medskip
  
\begin{itemize}
\item \textbf{\tr{Problem:}} \texttt{\textbackslash bibliography} inside \texttt{frame} environment puts all references on one slide.
  \begin{itemize}
  \item For large bibliographies, this can be unreadable.
  \item This was easy with Beamer, using the \texttt{allowframebreaks} option.
  \item But \texttt{ltx-talk} does not have a similar option.
  \end{itemize}

\bigskip

\item \textbf{\tg{Solution:}} Write code to split bibliography automatically across multiple slides
  \begin{itemize}
  \item Works with standard Bib\TeX\ workflow
  \item Uses Lua script (\texttt{\tb{rsbibsplit.lua}}) to split the \texttt{.bbl} file
  \item (Lua\LaTeX\ provides access to the \tb{Lua} programming language)
  \end{itemize}

\bigskip

\item \textbf{Usage:} As usual (see end of this \texttt{.tex} file for complete example)
  \begin{itemize}
  \item \texttt{\textbackslash cite\{key1,key2,...\}} to specify references
  \item \texttt{\textbackslash bibliography\{bibfile\}} to specify \texttt{bib} file
  \item (Set \texttt{\textbackslash RS@bibmax} to customize max per slide; default=8).
  \item \tb{\textbf{Note:} Do \textbf{not} put \texttt{\textbackslash bibliography} command inside \texttt{frame} environment }
    
  \end{itemize}

\bigskip

\item See \texttt{slide\_utils.sty} Section 2 for detailed workflow explanation.
\end{itemize}
\end{frame}
